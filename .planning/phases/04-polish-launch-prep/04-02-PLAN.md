    Optimize ModelList with memoization and FlashList best practices (mobile-app/components/ModelList.tsx):

    Add memoized data and renderItem:
    ```tsx
    import { FlashList } from '@shopify/flash-list';
    import { useCallback, useMemo } from 'react';

    interface ModelListProps {
      models: Model[];
      highlight?: string;
      refreshing?: boolean;
      onRefresh?: () => void;
    }

    export default function ModelList({ models, highlight, refreshing = false, onRefresh }: ModelListProps) {
      // Memoize data to prevent unnecessary re-renders
      const memoizedData = useMemo(() => models, [models]);

      // Optimize renderItem with useCallback to prevent new function on each render
      const renderItem = useCallback(({ item }: { item: Model }) => {
        return <ModelCard model={item} highlight={highlight} />;
      }, [highlight]);

      // Memoize keyExtractor for consistency
      const keyExtractor = useCallback((item: Model) => item.name, []);

      return (
        <FlashList
          data={memoizedData}
          renderItem={renderItem}
          keyExtractor={keyExtractor}
          estimatedItemSize={140} // Adjust based on actual ModelCard height
          refreshControl={
            <RefreshControl refreshing={refreshing} onRefresh={onRefresh} />
          }
          maintainVisibleContentPosition={{
            minIndexForVisible: 0,
            autoscrollToTopThreshold: 10,
          }}
          ListEmptyComponent={
            <EmptyMessage message="No models found matching your filters" />
          }
          removeClippedSubviews={true} // Remove off-screen views from memory
          windowSize={10} // Render 10 screens worth of content around viewport
          initialNumToRender={10} // Initial batch of items to render
          maxToRenderPerBatch={10} // Max items per batch during scroll
          updateCellsBatchingPeriod={50} // Frequency of cell updates (ms)
        />
      );
    }
    ```
